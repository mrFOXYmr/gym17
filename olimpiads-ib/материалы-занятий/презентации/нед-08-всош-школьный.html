    • Универсальная «раскрутка» (написать с нуля)

    import base64

def unwrap(s: str) -> str:
    while True:
        for fn in (base64.b64decode, base64.b32decode, base64.b16decode):
            try:
                s = fn(s.encode() if isinstance(s, str) else s).decode()
                break
            except Exception:
                continue
        else:
            break
    return s
